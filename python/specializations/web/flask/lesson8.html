أساسيات رفع الملفات

في Flask، يتم رفع الملفات باستخدام request.files. تحتاج النماذج إلى enctype="multipart/form-data".

نموذج رفع ملف

<form method="POST" action="/upload" enctype="multipart/form-data">
    <input type="file" name="file" accept=".jpg,.png,.pdf">
    <button type="submit">رفع الملف</button>
</form>
معالجة الملف في Flask

from flask import Flask, request, redirect, url_for
import os

app = Flask(__name__)
UPLOAD_FOLDER = 'uploads/'
ALLOWED_EXTENSIONS = {'png', 'jpg', 'jpeg', 'gif', 'pdf'}

app.config['UPLOAD_FOLDER'] = UPLOAD_FOLDER
app.config['MAX_CONTENT_LENGTH'] = 16 * 1024 * 1024  # 16MB

def allowed_file(filename):
    return '.' in filename and \
           filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS

@app.route('/upload', methods=['GET', 'POST'])
def upload_file():
    if request.method == 'POST':
        if 'file' not in request.files:
            return 'لا يوجد ملف', 400
        
        file = request.files['file']
        
        if file.filename == '':
            return 'لم يتم اختيار ملف', 400
        
        if file and allowed_file(file.filename):
            filename = secure_filename(file.filename)
            file.save(os.path.join(app.config['UPLOAD_FOLDER'], filename))
            return 'تم رفع الملف بنجاح!', 200
        
        return 'نوع الملف غير مسموح', 400
    
    return render_template('upload.html')
تنبيه مهم: استخدم secure_filename() لتأمين اسم الملف ومنع الهجمات. لا تثق أبداً في اسم الملف المقدم من المستخدم.

معالجة الصور باستخدام Pillow

Pillow هي مكتبة Python لمعالجة الصور. يمكنك استخدامها لتغيير الحجم، الضغط، والتحويل.

تثبيت Pillow

pip install Pillow
معالجة الصور

from PIL import Image
import os

def process_image(file_path, output_path, size=(800, 600)):
    """تغيير حجم الصورة وضغطها"""
    try:
        with Image.open(file_path) as img:
            # تغيير الحجم مع الحفاظ على النسبة
            img.thumbnail(size, Image.Resampling.LANCZOS)
            
            # تحويل إلى RGB إذا كانت RGBA (شفافية)
            if img.mode == 'RGBA':
                img = img.convert('RGB')
            
            # حفظ الصورة بجودة مخفضة
            img.save(output_path, 'JPEG', quality=85, optimize=True)
            return True
    except Exception as e:
        print(f'خطأ في معالجة الصورة: {e}')
        return False

# استخدام في دالة الرفع
def upload_and_process(file):
    if file and allowed_file(file.filename):
        filename = secure_filename(file.filename)
        raw_path = os.path.join(UPLOAD_FOLDER, 'raw', filename)
        processed_path = os.path.join(UPLOAD_FOLDER, 'processed', filename)
        
        # حفظ الملف الخام
        file.save(raw_path)
        
        # معالجة الصورة
        if filename.lower().endswith(('.jpg', '.jpeg', '.png', '.gif')):
            process_image(raw_path, processed_path, size=(800, 600))
            os.remove(raw_path)  # حذف الملف الخام
            return processed_path
        
        return raw_path
    return None
نصيحة: استخدم thumbnail() بدلاً من resize() للحفاظ على نسبة العرض إلى الارتفاع.

عرض الملفات

لعرض الملفات المرفوعة، يمكنك إنشاء نقاط نهاية ترجع الملفات أو تعرضها في HTML.

عرض الصور

from flask import send_from_directory

@app.route('/uploads/<filename>')
def uploaded_file(filename):
    """عرض الملفات المرفوعة"""
    return send_from_directory(app.config['UPLOAD_FOLDER'], filename)

@app.route('/images/<filename>')
def uploaded_image(filename):
    """عرض الصور مع تعيين نوع المحتوى"""
    return send_from_directory(
        app.config['UPLOAD_FOLDER'], 
        filename,
        mimetype='image/jpeg'
    )
عرض الصور في HTML

<!-- عرض صورة -->
<img src="{{ url_for('uploaded_image', filename='photo.jpg') }}" alt="صورة">

<!-- رابط تحميل ملف -->
<a href="{{ url_for('uploaded_file', filename='document.pdf') }}" download>
    تحميل الملف
</a>

<!-- عرض جميع الصور -->
{% for file in files %}
    <div>
        <img src="{{ url_for('uploaded_image', filename=file) }}" width="200">
        <p>{{ file }}</p>
    </div>
{% endfor %}

نصائح للتعامل مع الملفات

  • استخدم أسماء ملفات آمنة: استخدم secure_filename() أو أسماء عشوائية مثل UUID.
  • حدد الحجم الأقصى: استخدم MAX_CONTENT_LENGTH لمنع الملفات الكبيرة جداً.
  • تحقق من نوع الملف الفعلي: استخدم magic أو mimetypes بدلاً من الاعتماد على الامتداد فقط.
  • قم بضغط الصور: استخدم Pillow لتقليل حجم الصور وتحسين الأداء.
  • استخدم التخزين السحابي: للمشاريع الكبيرة، استخدم S3 أو Cloud Storage بدلاً من التخزين المحلي.
  • نظف الملفات غير المستخدمة: قم بحذف الملفات القديمة أو غير المستخدمة بشكل دوري.
نصيحة اليوم: "الملفات التي يرفعها المستخدمون هي بوابات محتملة للهجمات. عاملها دائماً بحذر وأمان."
