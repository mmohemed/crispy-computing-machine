نموذج المستخدم مع المصادقة

نموذج المستخدم يحتاج إلى دعم المصادقة وتجزئة كلمة المرور.

models.py - نموذج المستخدم

from flask_sqlalchemy import SQLAlchemy
from flask_login import UserMixin
from datetime import datetime

db = SQLAlchemy()

class User(db.Model, UserMixin):
    __tablename__ = 'users'
    
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(50), unique=True, nullable=False)
    email = db.Column(db.String(120), unique=True, nullable=False)
    password_hash = db.Column(db.String(128), nullable=False)  # تخزين التجزئة فقط
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    is_active = db.Column(db.Boolean, default=True)
    last_login = db.Column(db.DateTime, nullable=True)
    
    def __repr__(self):
        return f"User('{self.username}')"
    
    # دوال مساعدة للمصادقة
    def get_id(self):
        return str(self.id)
    
    def is_authenticated(self):
        return True
    
    def is_active_user(self):
        return self.is_active
    
    def is_anonymous(self):
        return False
تنبيه مهم: لا تقم أبداً بتخزين كلمات المرور كنص عادي في قاعدة البيانات. استخدم دائماً Bcrypt أو خوارزميات تجزئة قوية أخرى.

تجزئة كلمات المرور باستخدام Bcrypt

Bcrypt يقوم بتجزئة كلمة المرور مع إضافة "ملح" عشوائي، مما يجعلها آمنة جداً.

utils.py - دوال التجزئة

import bcrypt
from flask import current_app

def hash_password(password):
    """تجزئة كلمة المرور باستخدام Bcrypt"""
    # توليد ملح عشوائي
    salt = bcrypt.gensalt(rounds=12)
    # تجزئة كلمة المرور مع الملح
    hashed = bcrypt.hashpw(password.encode('utf-8'), salt)
    return hashed.decode('utf-8')

def verify_password(password, hashed):
    """التحقق من صحة كلمة المرور"""
    return bcrypt.checkpw(password.encode('utf-8'), hashed.encode('utf-8'))

# استخدام في نموذج المستخدم
def set_password(self, password):
    self.password_hash = hash_password(password)

def check_password(self, password):
    return verify_password(password, self.password_hash)
ملاحظة: bcrypt.checkpw() يقارن كلمة المرور المدخلة مع التجزئة المخزنة بأمان، دون الحاجة لتخزين الملح بشكل منفصل.

تكوين Flask-Login

Flask-Login يدير جلسات المستخدمين ويتذكر من هو المسجل دخوله.

app.py - تكوين Flask-Login

from flask import Flask
from flask_login import LoginManager
from models import db, User

app = Flask(__name__)
app.config['SECRET_KEY'] = 'your-secret-key-here'

# تهيئة قاعدة البيانات
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///site.db'
db.init_app(app)

# تهيئة Flask-Login
login_manager = LoginManager()
login_manager.init_app(app)
login_manager.login_view = 'login'  # المسار لصفحة تسجيل الدخول
login_manager.login_message = 'الرجاء تسجيل الدخول للوصول إلى هذه الصفحة'
login_manager.login_message_category = 'info'

@login_manager.user_loader
def load_user(user_id):
    """تحميل المستخدم من قاعدة البيانات باستخدام المعرف"""
    return User.query.get(int(user_id))
نصيحة: user_loader هو دالة مهمة جداً تستخدم لتحميل المستخدم من قاعدة البيانات لكل طلب.

نماذج تسجيل الدخول والتسجيل

استخدام Flask-WTF لإنشاء نماذج آمنة لتسجيل الدخول وإنشاء الحساب.

forms.py - نماذج المصادقة

from flask_wtf import FlaskForm
from wtforms import StringField, PasswordField, BooleanField, SubmitField
from wtforms.validators import DataRequired, Email, Length, EqualTo, ValidationError
from models import User

class RegistrationForm(FlaskForm):
    username = StringField('اسم المستخدم', 
        validators=[DataRequired(), Length(min=3, max=50)])
    
    email = StringField('البريد الإلكتروني', 
        validators=[DataRequired(), Email()])
    
    password = PasswordField('كلمة المرور', 
        validators=[DataRequired(), Length(min=6)])
    
    confirm_password = PasswordField('تأكيد كلمة المرور', 
        validators=[DataRequired(), EqualTo('password')])
    
    submit = SubmitField('إنشاء حساب')
    
    def validate_username(self, username):
        """التحقق من عدم وجود اسم مستخدم مكرر"""
        user = User.query.filter_by(username=username.data).first()
        if user:
            raise ValidationError('اسم المستخدم موجود بالفعل. اختر اسماً آخر.')
    
    def validate_email(self, email):
        """التحقق من عدم وجود بريد إلكتروني مكرر"""
        user = User.query.filter_by(email=email.data).first()
        if user:
            raise ValidationError('البريد الإلكتروني موجود بالفعل. استخدم بريداً آخر.')

class LoginForm(FlaskForm):
    email = StringField('البريد الإلكتروني', 
        validators=[DataRequired(), Email()])
    
    password = PasswordField('كلمة المرور', 
        validators=[DataRequired()])
    
    remember = BooleanField('تذكرني')
    submit = SubmitField('تسجيل الدخول')

دوال تسجيل الدخول والخروج

تنفيذ دوال المصادقة الأساسية باستخدام Flask-Login.

routes.py - دوال المصادقة

from flask import render_template, redirect, url_for, flash, request
from flask_login import login_user, logout_user, current_user, login_required
from models import User, db
from forms import RegistrationForm, LoginForm

@app.route('/register', methods=['GET', 'POST'])
def register():
    if current_user.is_authenticated:
        return redirect(url_for('home'))
    
    form = RegistrationForm()
    if form.validate_on_submit():
        # إنشاء مستخدم جديد مع تجزئة كلمة المرور
        user = User(
            username=form.username.data,
            email=form.email.data,
            password_hash=hash_password(form.password.data)
        )
        db.session.add(user)
        db.session.commit()
        
        flash('تم إنشاء حسابك بنجاح! يمكنك الآن تسجيل الدخول.', 'success')
        return redirect(url_for('login'))
    
    return render_template('register.html', form=form)

@app.route('/login', methods=['GET', 'POST'])
def login():
    if current_user.is_authenticated:
        return redirect(url_for('home'))
    
    form = LoginForm()
    if form.validate_on_submit():
        user = User.query.filter_by(email=form.email.data).first()
        
        if user and verify_password(form.password.data, user.password_hash):
            # تسجيل دخول المستخدم
            login_user(user, remember=form.remember.data)
            user.last_login = datetime.utcnow()
            db.session.commit()
            
            # إعادة التوجيه إلى الصفحة التي كان يريدها
            next_page = request.args.get('next')
            flash('تم تسجيل الدخول بنجاح!', 'success')
            return redirect(next_page) if next_page else redirect(url_for('home'))
        else:
            flash('فشل تسجيل الدخول. تحقق من البريد الإلكتروني وكلمة المرور.', 'danger')
    
    return render_template('login.html', form=form)

@app.route('/logout')
@login_required
def logout():
    logout_user()
    flash('تم تسجيل الخروج بنجاح.', 'info')
    return redirect(url_for('home'))

حماية المسارات

استخدام @login_required لحماية المسارات التي تتطلب تسجيل الدخول.

حماية المسارات

from flask_login import login_required, current_user

@app.route('/dashboard')
@login_required
def dashboard():
    """صفحة لوحة التحكم - فقط للمستخدمين المسجلين"""
    return render_template('dashboard.html', user=current_user)

@app.route('/profile')
@login_required
def profile():
    """صفحة الملف الشخصي - فقط للمستخدمين المسجلين"""
    return render_template('profile.html', user=current_user)

@app.route('/admin')
@login_required
def admin():
    """صفحة المدير - فقط للمستخدمين مع صلاحيات خاصة"""
    if not current_user.is_admin:  # صلاحية إضافية
        flash('ليس لديك صلاحية الوصول إلى هذه الصفحة.', 'danger')
        return redirect(url_for('home'))
    return render_template('admin.html')
ملاحظة: @login_required يعيد توجيه المستخدم تلقائياً إلى صفحة تسجيل الدخول مع حفظ عنوان الصفحة المطلوبة عبر next.

محاكي المصادقة التفاعلي

جرب نظام المصادقة الآمن في Flask من خلال هذا المحاكي التفاعلي.

غير مسجل دخول
🔒 حالة الجلسة: لا توجد جلسة نشطة
تنبيه: هذا المحاكي تفاعلي فقط ويوضح آلية عمل المصادقة. البيانات لا تُحفظ بشكل دائم.

نصائح للمصادقة الآمنة

  • استخدم تجزئة قوية: Bcrypt هو الخيار الموصى به لتجزئة كلمات المرور.
  • طبق حداً لمحاولات الدخول: امنع المستخدمين بعد عدة محاولات فاشلة لمنع هجمات القوة العمياء.
  • استخدم التحقق بخطوتين: أضف طبقة أمان إضافية باستخدام رمز التحقق (2FA).
  • سجل الخروج التلقائي: قم بتسجيل خروج المستخدمين تلقائياً بعد فترة من عدم النشاط.
  • حمِ من هجمات CSRF: استخدم CSRFProtect مع جميع النماذج التي ترسل بيانات حساسة.
نصيحة اليوم: "الأمن ليس ميزة إضافية، بل هو جزء أساسي من عملية التطوير. ابنِ الأمان من البداية."
